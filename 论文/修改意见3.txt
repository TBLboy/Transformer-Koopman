    \caption{Overall prediction pipeline of the proposed
    Transformer-Koopman framework. Given a history window $\mX_k$ and
    the current control input $\vu_k$, the Transformer encoder extracts
    a history feature $\vh_k$, which is concatenated with the current
    state $\vx_k$ to form the encoder-consistent latent
    $\vz^{E}_k = [\vx_k;\,\vh_k] \in \mathbb{R}^d$, while the context
    vector $\vc_k = C_\phi(\mX_k,\mU^{\mathrm{in}}_k)$ additionally
    consumes the past inputs. The Koopman linear dynamics
    $\hat{\vz}^{P}_{k+1} = \mA(\vc_k)\vz^{E}_k + \mB\vu_k$ propagate this
    latent forward, and the predicted next physical state
    $\hat{\vx}_{k+1} = \mC\hat{\vz}^{P}_{k+1}$ is recovered through a
    fixed linear projection; the superscripts $E$ and $P$ distinguish the
    encoder-consistent latent from the model-propagated latent.}
    \label{fig:framework}
\end{figure*}
