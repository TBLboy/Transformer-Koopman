% =========================================================
\section{Controller Design and Stability Analysis}
\label{sec:control}
% =========================================================

In this section, we develop a tracking controller based on the identified Koopman model and analyze the resulting closed-loop system.

% ---------------------------------------------------------
\subsection{Controller Design}
\label{subsec:controller_design}
% ---------------------------------------------------------

Based on the identified lifted dynamics \eqref{eq:temporal_dynamics} and the structured lifted state \eqref{eq:lifted_state_form}, we design a tracking controller in the lifted space. Let $\{x_k^{\mathrm{ref}}\}$ denote a reference state trajectory, and define the corresponding lifted reference trajectory as
\begin{equation}
    z_k^{\mathrm{ref}} = [x_k^{\mathrm{ref}};\,h_k^{\mathrm{ref}}],
    \label{eq:lifted_reference}
\end{equation}
where $h_k^{\mathrm{ref}}$ is obtained by encoding the reference history window. The lifted tracking error is then defined as
\begin{equation}
    e_k^z = z_k - z_k^{\mathrm{ref}}
    =
    \begin{bmatrix}
        e_k^x \\ e_k^h
    \end{bmatrix}
    =
    \begin{bmatrix}
        x_k - x_k^{\mathrm{ref}} \\
        h_k - h_k^{\mathrm{ref}}
    \end{bmatrix}.
    \label{eq:structured_error}
\end{equation}

We decompose the control input into feedforward and feedback terms,
\begin{equation}
    u_k = u_{\mathrm{ff},k} + u_{\mathrm{fb},k}.
    \label{eq:control_decomposition}
\end{equation}
The feedforward term compensates for the nominal reference evolution in the lifted space and is chosen as
\begin{equation}
    u_{\mathrm{ff},k}
    =
    B^\dagger \!\left(z_{k+1}^{\mathrm{ref}} - A z_k^{\mathrm{ref}}\right),
    \label{eq:feedforward}
\end{equation}
where $B^\dagger$ denotes the Moore--Penrose pseudoinverse of $B$.

The feedback term is designed from the structured lifted error as
\begin{equation}
    u_{\mathrm{fb},k}
    =
    -K_x e_k^x - K_h e_k^h
    =
    -K_c e_k^z,
    \label{eq:hierarchical_fb}
\end{equation}
where $K_c = [K_x\;|\;K_h]$ is partitioned according to \eqref{eq:structured_error}. To determine $K_c$, we perform a discrete-time LQR design for the pair $(A,B)$ with cost
\begin{equation}
    J = \sum_{k=0}^{\infty}
    \left[
        (e_k^z)^\top Q\,e_k^z
        +
        (u_{\mathrm{fb},k})^\top R\,u_{\mathrm{fb},k}
    \right],
    \label{eq:lqr_objective}
\end{equation}
where $R \succ 0$ and
\begin{equation}
    Q =
    \begin{bmatrix}
        Q_x & 0 \\
        0 & \alpha Q_h
    \end{bmatrix},
    \qquad \alpha \in [0,1).
    \label{eq:block_diag_Q}
\end{equation}
Here, $Q_x$ determines the primary tracking objective in the physical state, whereas $\alpha Q_h$ assigns a smaller auxiliary penalty to the feature error. In practice, $Q_x$ is chosen dominant and $\alpha$ is set small, with $\alpha = 0$ also admissible if no explicit penalty on $e_k^h$ is desired. Solving the standard discrete-time Riccati equation yields
\begin{equation}
    K_c
    =
    (R + B^\top P B)^{-1} B^\top P A
    =
    [K_x\;|\;K_h].
    \label{eq:lqr_gain}
\end{equation}

Combining \eqref{eq:control_decomposition}--\eqref{eq:hierarchical_fb}, the resulting control law is
\begin{equation}
    u_k
    =
    B^\dagger \!\left(z_{k+1}^{\mathrm{ref}} - A z_k^{\mathrm{ref}}\right)
    - K_c e_k^z.
    \label{eq:full_ctrl}
\end{equation}

The overall controller structure is illustrated in Fig.~\ref{fig:control_architecture}.

\begin{figure}[!t]
    \centering
    %\includegraphics[width=0.9\linewidth]{figs/fig2_control_architecture.pdf}
    \caption{Control architecture based on the structured lifted state $z_k=[x_k;h_k]$. The reference trajectory is encoded to $z_k^{\mathrm{ref}}=[x_k^{\mathrm{ref}};h_k^{\mathrm{ref}}]$, and the current state history is encoded to $z_k=[x_k;h_k]$. The error $e_k^z = z_k - z_k^{\mathrm{ref}}$ drives both feedforward and feedback control terms.}

% ---------------------------------------------------------
\subsection{Stability Analysis}
\label{subsec:stability}
% ---------------------------------------------------------

We analyze the closed-loop tracking error induced by the control law~\eqref{eq:full_ctrl} applied to the identified Koopman model.

\begin{theorem}[Ultimate Boundedness of Tracking Error]
\label{thm:ub}
Consider the lifted dynamics~\eqref{eq:temporal_dynamics} under the control law~\eqref{eq:full_ctrl}. Suppose that $(A, B)$ is stabilizable, $(A, Q^{1/2})$ is detectable, and the modeling residual satisfies $\|w_k\| \leq \varepsilon$ for all $k \geq 0$. Then the closed-loop system achieves ultimately uniformly bounded tracking, i.e., the lifted tracking error satisfies
\begin{equation}
    \|e_k^z\| \leq \frac{\varepsilon}{1 - \|A_{\mathrm{cl}}\|}
    \label{eq:ub_bound}
\end{equation}
for all sufficiently large $k$, and consequently
\begin{equation}
    \|x_k - x_k^{\mathrm{ref}}\| \leq \frac{\varepsilon}{1 - \|A_{\mathrm{cl}}\|}.
    \label{eq:phys_bound}
\end{equation}
\end{theorem}

\begin{proof}
Substituting~\eqref{eq:full_ctrl} into~\eqref{eq:temporal_dynamics} and using~\eqref{eq:modeling_error}, the tracking error $e_k^z = z_k - z_k^{\mathrm{ref}}$ evolves as
\begin{equation}
    e_{k+1}^z = A_{\mathrm{cl}}\, e_k^z + w_k,
    \label{eq:cl_error_dyn}
\end{equation}
where $A_{\mathrm{cl}} = A - BK_c$. By the stabilizability and detectability assumptions and standard LQR theory, $A_{\mathrm{cl}}$ is Schur stable with $\|A_{\mathrm{cl}}\| < 1$. Iterating~\eqref{eq:cl_error_dyn} and applying the triangle inequality, we obtain
\begin{equation}
    e_k^z
    =
    A_{\mathrm{cl}}^k e_0^z
    +
    \sum_{i=0}^{k-1}
    A_{\mathrm{cl}}^{k-1-i} w_i.
\end{equation}
Taking norms and using $\|w_i\| \leq \varepsilon$ gives
\begin{equation}
    \|e_k^z\|
    \leq
    \|A_{\mathrm{cl}}\|^k \|e_0^z\|
    +
    \varepsilon \sum_{i=0}^{k-1} \|A_{\mathrm{cl}}\|^i.
\end{equation}
Since $\|A_{\mathrm{cl}}\| < 1$, the geometric series is bounded, and hence
\begin{equation}
    \limsup_{k\to\infty}\|e_k^z\|
    \leq
    \frac{\varepsilon}{1-\|A_{\mathrm{cl}}\|},
\end{equation}
which proves \eqref{eq:ub_bound}. Because the physical state is recovered from the lifted state by $x_k = Cz_k$ with $\|C\| = 1$, one further has
\begin{equation}
    \|x_k - x_k^{\mathrm{ref}}\|
    =
    \|C e_k^z\|
    \leq
    \|e_k^z\|,
\end{equation}
which yields \eqref{eq:phys_bound}.
\end{proof}

The ultimate bound scales linearly with the modeling residual $\varepsilon$: improving the Koopman approximation accuracy directly tightens the closed-loop tracking guarantee, and as $\varepsilon \to 0$, the tracking error converges to zero asymptotically.
